# NOTE
# [c_block_terms]
#   DAP: Disk Address Packet
#   LBA: Logical Bolck Address
#
# [n_set_dap]
#   [dap+2] sector count
#   [dap+4] offset
#   [dap+6] segment
#   [dap+8] lba low
#   [dap+10] lba high
#
# [n_rw_block]
#   mode: 0x42 (read), 0x43 (write)
#
# [n_dap]
#   Common block
#     sector count: 8
#     mem ptr: 0x8000
#     lba low addr: 0x80-0xFFFF
#     lba high addr: 0x00-0xFFFF
#       fayos uses only 4-byte lba
# 
#   Super block
#     sector count: 4
#     mem ptr: 0x0600
#     lba addr: 0x10, 0x10-0x13
