# INDEX
# set_dentry()

# NOTE
# [c_dentry]
#   [dentry_variable]
#     [idx-((name_size)+(padding_size))] name: 1-242 bytes (256 - [dentry_fixed])
#     [idx-1] name_align: 0-1 byte
#
#   [dentry_fixed]
#     [idx+0] magic_num: 2 bytes (0xFADE: FacooyA Directory Entry)
#     [idx+2] name_size: 1 byte (for name)
#     [idx+3] padding_size: 1 byte (for name_align)
#     [idx+4] data_lba: 4 bytes (total)
#       [idx+4] data_lba_low: 2 bytes (part of data_lba)
#       [idx+6] data_lba_high: 2 bytes (part of data_lba)
#     [idx+8] entry_level: 1 byte
#     [idx+9] file_type: 1 byte
#   
#     (more: time)
#
#
# [c_file_type]
#   0x0D: dir, 0x0E: exec, 0x0F: file 
#   MSB 1 is deleted. file_type << 7 == 1 ? deleted
#   E.g., 0x0F (file) + 0x80 = 0x8F (deleted file)
# 
# [n_write_dentry]
#   align: name_size % 2 = DL, SI += DL
#     SI: mem ptr (magic num)
#     padding size: 3(SI) = DL
